# Native upstream wimlib plus the selected-reader extension. No configure-time downloads.
set(RUFUS_WIMLIB_UPSTREAM "" CACHE PATH "Upstream wimlib 1.14.5 release source tree")
if(NOT EXISTS "${RUFUS_WIMLIB_UPSTREAM}/configure.ac")
    message(FATAL_ERROR "Set RUFUS_WIMLIB_UPSTREAM to the pinned source described in docs/linux-wimlib-prototype.md")
endif()
find_package(Threads REQUIRED)
find_program(PATCH_PROGRAM patch REQUIRED)
# Applying a Windows installation needs NTFS security descriptors, streams and
# reparse points. The Unix directory backend cannot preserve those metadata.
# This is an offline library operation, independent of our kernel/FUSE mounts.
find_path(RUFUS_WIM_NTFS_PLUS_INCLUDE NAMES ntfsprogs-plus/attrib.h)
find_library(RUFUS_WIM_NTFS_PLUS_LIBRARY NAMES ntfs)
if(RUFUS_WIM_NTFS_PLUS_INCLUDE AND RUFUS_WIM_NTFS_PLUS_LIBRARY)
    set(wim_ntfs_include "${RUFUS_WIM_NTFS_PLUS_INCLUDE}/ntfsprogs-plus")
    set(wim_ntfs_library "${RUFUS_WIM_NTFS_PLUS_LIBRARY}")
    set(wim_ntfs_provider "ntfsprogs-plus")
else()
    find_path(RUFUS_WIM_NTFS_3G_INCLUDE NAMES ntfs-3g/attrib.h REQUIRED)
    find_library(RUFUS_WIM_NTFS_3G_LIBRARY NAMES ntfs-3g REQUIRED)
    set(wim_ntfs_include "${RUFUS_WIM_NTFS_3G_INCLUDE}/ntfs-3g")
    set(wim_ntfs_library "${RUFUS_WIM_NTFS_3G_LIBRARY}")
    set(wim_ntfs_provider "ntfs-3g")
endif()
message(STATUS "Native WIM NTFS provider: ${wim_ntfs_provider}")
set(RUFUS_WIM_NTFS_PROVIDER "${wim_ntfs_provider}" PARENT_SCOPE)
add_library(rufus_native_ntfs UNKNOWN IMPORTED GLOBAL)
set_target_properties(rufus_native_ntfs PROPERTIES IMPORTED_LOCATION "${wim_ntfs_library}"
    INTERFACE_INCLUDE_DIRECTORIES "${wim_ntfs_include}")
include(CheckIncludeFile)
include(CheckSymbolExists)
include(CheckStructHasMember)
include(CheckTypeSize)
set(stage "${CMAKE_CURRENT_BINARY_DIR}/source")
set(CMAKE_REQUIRED_DEFINITIONS -D_GNU_SOURCE -D_FILE_OFFSET_BITS=64)
foreach(header alloca.h byteswap.h endian.h errno.h glob.h inttypes.h stdarg.h
        stddef.h stdint.h stdio.h stdlib.h strings.h string.h sys/file.h
        sys/stat.h sys/syscall.h sys/times.h sys/types.h sys/xattr.h time.h unistd.h utime.h)
    string(MAKE_C_IDENTIFIER "HAVE_${header}" feature)
    string(TOUPPER "${feature}" feature)
    check_include_file("${header}" ${feature})
endforeach()
foreach(symbol futimens utimensat flock mempcpy openat fstatat readlinkat
        fdopendir posix_fallocate llistxattr lgetxattr fsetxattr lsetxattr)
    string(TOUPPER "HAVE_${symbol}" feature)
    check_symbol_exists(${symbol}
        "stdio.h;stdlib.h;string.h;sys/stat.h;sys/file.h;fcntl.h;unistd.h;dirent.h;sys/xattr.h" ${feature})
endforeach()
check_struct_has_member("struct stat" st_mtim "sys/stat.h" HAVE_STAT_NANOSECOND_PRECISION)
set(CMAKE_EXTRA_INCLUDE_FILES sys/types.h)
check_type_size(off_t WIM_OFF_T_SIZE LANGUAGE C)
unset(CMAKE_EXTRA_INCLUDE_FILES)
unset(CMAKE_REQUIRED_DEFINITIONS)
if(NOT WIM_OFF_T_SIZE EQUAL 8)
    message(FATAL_ERROR "Native wimlib requires 64-bit off_t")
endif()
if(CMAKE_C_BYTE_ORDER STREQUAL "BIG_ENDIAN")
    set(WORDS_BIGENDIAN 1)
elseif(NOT CMAKE_C_BYTE_ORDER STREQUAL "LITTLE_ENDIAN")
    message(FATAL_ERROR "Unknown wimlib byte order")
endif()

# Verify every consumed upstream source/header, then patch only build copies.
# configure_file causes Ninja to reconfigure after source edits.
file(READ upstream-inputs.json upstream_inputs)
string(JSON input_count LENGTH "${upstream_inputs}")
math(EXPR last_input "${input_count} - 1")
foreach(index RANGE ${last_input})
    string(JSON name MEMBER "${upstream_inputs}" ${index})
    string(JSON expected GET "${upstream_inputs}" "${name}")
    set(input "${RUFUS_WIMLIB_UPSTREAM}/${name}")
    file(SHA256 "${input}" actual)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "Wrong upstream wimlib source: ${name}")
    endif()
    string(REGEX REPLACE "^(src|include)/" "" output "${name}")
    configure_file("${input}" "${stage}/${output}" COPYONLY)
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    native-input.patch native-ntfs.patch upstream-inputs.json)
execute_process(COMMAND "${PATCH_PROGRAM}" --batch --fuzz=0 -p1 -i "${CMAKE_CURRENT_SOURCE_DIR}/native-input.patch"
    WORKING_DIRECTORY "${stage}" RESULT_VARIABLE patch_result OUTPUT_VARIABLE patch_output ERROR_VARIABLE patch_error)
if(NOT patch_result EQUAL 0)
    message(FATAL_ERROR "Native wimlib patch failed: ${patch_output}${patch_error}")
endif()
execute_process(COMMAND "${PATCH_PROGRAM}" --batch --fuzz=0 -p1 -i "${CMAKE_CURRENT_SOURCE_DIR}/native-ntfs.patch"
    WORKING_DIRECTORY "${stage}" RESULT_VARIABLE patch_result OUTPUT_VARIABLE patch_output ERROR_VARIABLE patch_error)
if(NOT patch_result EQUAL 0)
    message(FATAL_ERROR "Native NTFS patch failed: ${patch_output}${patch_error}")
endif()
# Both providers expose the same headers under different directory names.
# Keep the verified upstream algorithms and select only the include directory.
foreach(name ntfs-3g_apply.c ntfs-3g_capture.c)
    file(READ "${stage}/${name}" ntfs_source)
    string(REPLACE "#include <ntfs-3g/" "#include <" ntfs_source "${ntfs_source}")
    file(WRITE "${stage}/${name}" "${ntfs_source}")
endforeach()
configure_file(config.h.in "${stage}/config.h")
include(sources.cmake)
list(TRANSFORM wim_sources PREPEND "${stage}/")
add_library(rufus_wimlib STATIC ${wim_sources} native_input.c native_ntfs_ea.c native_ntfs_efs.c)
target_compile_features(rufus_wimlib PRIVATE c_std_11)
target_compile_definitions(rufus_wimlib PRIVATE
    HAVE_CONFIG_H=1 BUILDING_WIMLIB=1 _GNU_SOURCE _LARGEFILE_SOURCE _FILE_OFFSET_BITS=64)
target_compile_options(rufus_wimlib PRIVATE -Wall -Wextra -Wno-pointer-sign)
target_include_directories(rufus_wimlib PUBLIC "${stage}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(rufus_wimlib PUBLIC rufus_iso_scan PRIVATE Threads::Threads rufus_native_ntfs)
