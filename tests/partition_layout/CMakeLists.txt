cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(RufusPartitionLayoutTests LANGUAGES C)
    include(CTest)
endif()
add_executable(test_partition_layout test_partition_layout.c ../../src/partition_layout.c)
target_include_directories(test_partition_layout PRIVATE ../../src)
target_compile_features(test_partition_layout PRIVATE c_std_11)
if(MSVC)
    target_compile_options(test_partition_layout PRIVATE /W4)
else()
    target_compile_options(test_partition_layout PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME rufus_partition_layout COMMAND test_partition_layout)
set_tests_properties(rufus_partition_layout PROPERTIES LABELS "component;image;no-hardware" TIMEOUT 10)
