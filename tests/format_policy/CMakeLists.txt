cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(RufusFormatPolicyTests LANGUAGES C)
    include(CTest)
endif()
add_executable(test_format_policy test_format_policy.c boot_dispatch_reference.c ../../src/format_policy.c)
target_include_directories(test_format_policy PRIVATE ../../src)
target_compile_features(test_format_policy PRIVATE c_std_11)
if(MSVC)
    target_compile_options(test_format_policy PRIVATE /W4)
    target_compile_definitions(test_format_policy PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(test_format_policy PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(test_format_policy PRIVATE m)
endif()
add_test(NAME rufus_format_policy COMMAND test_format_policy)
set_tests_properties(rufus_format_policy PROPERTIES LABELS "component;image;no-hardware" TIMEOUT 10)
