cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(RufusUnattendTests LANGUAGES C)
    include(CTest)
endif()

if(NOT TARGET rufus_unattend)
    add_library(rufus_unattend STATIC ../../src/unattend.c ../../src/strarray.c)
    target_include_directories(rufus_unattend PUBLIC ../../src)
    target_compile_features(rufus_unattend PRIVATE c_std_11)
    if(MSVC)
        target_compile_options(rufus_unattend PRIVATE /W4 /utf-8)
        target_compile_definitions(rufus_unattend PRIVATE _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_options(rufus_unattend PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    if(NOT WIN32)
        target_compile_definitions(rufus_unattend PRIVATE _GNU_SOURCE)
        target_compile_options(rufus_unattend PRIVATE
            -include "${CMAKE_CURRENT_SOURCE_DIR}/../../linux/platform/crt_strings.h")
    endif()
endif()

add_executable(test_unattend test_unattend.c)
target_link_libraries(test_unattend PRIVATE rufus_unattend)
target_compile_features(test_unattend PRIVATE c_std_11)
if(MSVC)
    target_compile_options(test_unattend PRIVATE /W4 /utf-8)
    target_compile_definitions(test_unattend PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(test_unattend PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME rufus_unattend COMMAND test_unattend)
set_tests_properties(rufus_unattend PROPERTIES LABELS "component;image;no-hardware" TIMEOUT 10)

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(test_unattend_failure test_unattend_failure.c)
    target_link_libraries(test_unattend_failure PRIVATE rufus_unattend)
    target_compile_options(test_unattend_failure PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(test_unattend_failure PRIVATE
        -Wl,--wrap=calloc -Wl,--wrap=realloc -Wl,--wrap=strdup -Wl,--wrap=ftell)
    add_test(NAME rufus_unattend_failure COMMAND test_unattend_failure)
    set_tests_properties(rufus_unattend_failure PROPERTIES LABELS "component;image;no-hardware" TIMEOUT 10)
endif()
