cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(RufusPeArchTests LANGUAGES C)
    include(CTest)
endif()
add_executable(test_pe_arch test_pe_arch.c ../../src/pe_arch.c)
target_include_directories(test_pe_arch PRIVATE ../../src)
target_compile_features(test_pe_arch PRIVATE c_std_11)
if(MSVC)
    target_compile_options(test_pe_arch PRIVATE /W4)
else()
    target_compile_options(test_pe_arch PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME rufus_pe_arch COMMAND test_pe_arch)
set_tests_properties(rufus_pe_arch PROPERTIES LABELS "component;image;no-hardware" TIMEOUT 10)
