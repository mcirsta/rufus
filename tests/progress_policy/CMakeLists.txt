cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(RufusProgressPolicyTests LANGUAGES C)
    include(CTest)
endif()
add_executable(test_progress_policy test_progress_policy.c ../../src/progress_policy.c)
target_include_directories(test_progress_policy PRIVATE ../../src)
target_compile_features(test_progress_policy PRIVATE c_std_11)
if(MSVC)
    target_compile_options(test_progress_policy PRIVATE /W4)
    target_compile_definitions(test_progress_policy PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(test_progress_policy PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(test_progress_policy PRIVATE m)
endif()
add_test(NAME rufus_progress_policy COMMAND test_progress_policy)
set_tests_properties(rufus_progress_policy PROPERTIES LABELS "component;image;no-hardware" TIMEOUT 10)
